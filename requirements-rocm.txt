# Full fine-tuning on AMD GPUs (scripts/train_full.py), e.g. 8x MI355X (gfx950).
# Install JAX for ROCm first, per AMD's JAX-on-ROCm docs for your ROCm version;
# for ROCm 7:
#   pip install jax jax-rocm7-plugin      # pulls the matching jax-rocm7-pjrt
# jax is unpinned here: the TPU-only splash-attention path (pinned in
# requirements-tpu.txt) is not used on GPU (--attn ref / aiter).
#
# Optional: jax-aiter (fused flash attention for --attn aiter, MI355X/gfx950
# only). Not on PyPI; it supports exactly one stack, so install it by hand:
#   image  ghcr.io/rocm/jax-base-ubu24.therock-7.14:7.14   (TheRock 7.14 GA)
#   pip install "jax==0.11.0" "jaxlib==0.11.0" \
#     "jax-rocm7-plugin==0.11.0.post1" "jax-rocm7-pjrt==0.11.0.post1"
#   pip install ./jax_aiter-0.1.0a2+full-cp312-cp312-manylinux_2_39_x86_64.whl
#   python -c "from jax_aiter.mha import flash_attn_varlen"
# There is no GitHub release for it. The +full wheel (MHA libs bundled, ~511
# MB) is the jax-aiter `nightly-wheel` GitHub Actions artifact (download needs
# `gh auth`); our runs use a mirror on the public HF dataset
# ricecake/krea2-ops-assets. The lite 0.1.0a2 wheel has no MHA libs; after
# installing it run
#   jax-aiter-fetch-mha --tag jit-libs-85d614f12c412443
# The --tag is required: the wheel's own RELEASE_TAG points at a manifest
# whose aiter_sha does not match the wheel (symptom: "named symbol not found"
# at the first aiter call, after the base load and compile). Python 3.12.
# Everything else in this repo runs without it.
jax

# on the next install, so the optimizer/checkpoint stack must not drift
optax==0.2.8
